SELECT f.tenant_id, su.id AS social_unit_id, s.code AS service_code, COUNT(*) AS qty
FROM family_follow_ups f
JOIN social_units su       ON su.id = f.social_unit_id
JOIN follow_up_services s  ON s.id  = f.follow_up_service_id
WHERE f.deleted_at IS NULL
  AND f.status <> 'cancelled'
  AND s.social_unit_type_id <> su.type_id
GROUP BY f.tenant_id, su.id, s.code;
