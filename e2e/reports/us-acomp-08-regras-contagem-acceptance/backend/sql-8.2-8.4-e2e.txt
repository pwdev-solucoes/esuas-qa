# SQL pronto (com o cast ::jsonb — activity_log.properties e json):
-- §8.2 — Servicos cujo tipo foi reapontado depois de ter acompanhamentos (V5)
-- NOTA: activity_log.properties e `json` neste projeto (migration 2026_05_21_185309), nao `jsonb`;
-- o operador `?` e o `->`/`->>` encadeados exigem o cast `::jsonb`. Semantica identica a da spec.
SELECT s.code, a.created_at,
       (a.properties::jsonb)->'old'->>'social_unit_type_id'        AS tipo_anterior,
       (a.properties::jsonb)->'attributes'->>'social_unit_type_id' AS tipo_novo
FROM activity_log a
JOIN follow_up_services s ON s.id = a.subject_id
WHERE a.subject_type = 'App\Models\FollowUpService'
  AND (a.properties::jsonb)->'attributes' ? 'social_unit_type_id'
  AND EXISTS (SELECT 1 FROM family_follow_ups f
              WHERE f.follow_up_service_id = s.id AND f.created_at < a.created_at);

-- §8.3 — Unidades com tipo alterado depois de ter acompanhamentos (V6)
SELECT su.tenant_id, su.id, a.created_at
FROM activity_log a
JOIN social_units su ON su.id = a.subject_id
WHERE a.subject_type = 'App\Models\SocialUnit'
  AND (a.properties::jsonb)->'attributes' ? 'type_id'
  AND EXISTS (SELECT 1 FROM family_follow_ups f
              WHERE f.social_unit_id = su.id AND f.created_at < a.created_at);

-- §8.4 — Frequencia de ingresso-desligamento-reingresso no mesmo mes (Duvida 3 do PO)
SELECT x.tenant_id, x.mes, COUNT(*) AS familias_com_reingresso_no_mes
FROM (
  SELECT f.tenant_id, f.social_unit_id, f.follow_up_service_id, f.family_id,
         date_trunc('month', f.admitted_on) AS mes
  FROM family_follow_ups f
  WHERE f.deleted_at IS NULL
    AND f.status <> 'cancelled'
  GROUP BY f.tenant_id, f.social_unit_id, f.follow_up_service_id, f.family_id, date_trunc('month', f.admitted_on)
  HAVING COUNT(*) > 1
) x
GROUP BY x.tenant_id, x.mes
ORDER BY x.mes;
